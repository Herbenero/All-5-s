} } class Player { final String name; int totalScore = 0; List currentRoundDarts = []; Player(this.name); // Deep copy for history snapshotting Player copy() { return Player(name) ..totalScore = totalScore ..currentRoundDarts = List.from(currentRoundDarts); } } class GameStateSnapshot { final List players; final int currentPlayerIndex; final String multiplier; GameStateSnapshot({ required this.players, required this.currentPlayerIndex, required this.multiplier, }); } } class ScoreboardScreen extends StatefulWidget { final List players; const ScoreboardScreen({super.key, required this.players}); @override State createState() => _ScoreboardScreenState(); } class _ScoreboardScreenState extends State { int currentPlayerIndex = 0; } else { p.currentRoundDarts.clear(); currentPlayerIndex = (currentPlayerIndex + 1) % widget.players.length; } } }); } void _showWinner(String name) { showDialog( context: context, barrierDismissible: false, builder: (context) => AlertDialog( title: const Text('🎉 Winner! 🎉', textAlign: centerTitle ? Colors.deepPurple.shade900 : null, shape: widget.players[currentPlayerIndex] == p ? RoundedRectangleBorder(side: BorderSide(color: Theme.of(context).colorScheme.primary, width: 2), borderRadius: BorderRadius.circular(12)) : null, child: Padding( padding: const EdgeInsets.all(12.0), child: Column( mainAxisAlignment: MainAxisAlignment.center, children: [ Text(p.name, style: const TextStyle(fontWeight: FontWeight.bold)), Text('Score: ${p.totalScore}/51', style: TextStyle(color: Theme.of(context).colorScheme.secondary)), ], ), ), )).toList(), ), ), Padding( padding: const EdgeInsets.all(8.0), child: Column( children: [ Text("Current Turn: ${current.name}", style: const TextStyle(fontSize: 20, fontWeight: FontWeight.bold)), const SizedBox(height: 4), Text( "Darts This Round: [ ${current.currentRoundDarts.join(', ')} ] (${current.currentRoundDarts.length}/3)", style: const TextStyle(fontSize: 16, color: Colors.grey), ), ], ), ), const Divider(), // Multiplier Selectors Row( mainAxisAlignment: MainAxisAlignment.spaceEvenly, children: ['Single', 'Double', 'Triple'].map((m) => ChoiceChip( label: Text(m), selected: multiplier == m, onSelected: (selected) { if (selected) setState(() => multiplier = m); }, )).toList(), ), const SizedBox(height: 8), // Core Targets Layout Matrix Expanded( child: GridView.count( crossAxisCount: 5, padding: const EdgeInsets.all(8), mainAxisSpacing: 6, crossAxisSpacing: 6, children: [ ...List.generate(20, (i) => i + 1).map((i) => ElevatedButton( style: ElevatedButton.styleFrom(padding: EdgeInsets.zero), onPressed: () => _handleScore(i), child: Text('$i', style: const TextStyle(fontSize: 16, fontWeight: FontWeight.bold)), )), ElevatedButton( style: ElevatedButton.styleFrom(backgroundColor: Colors.red.shade900), onPressed: () => _handleScore(25), child: const Text('BULL', style: TextStyle(color: Colors.white, fontWeight: FontWeight.bold)), ), ElevatedButton( style: ElevatedButton.styleFrom(backgroundColor: Colors.blueGrey.shade800), onPressed: () => _handleScore(0, isMiss: true), child: const Text('MISS', style: TextStyle(color: Colors.white)), ), ], ), ), // Control Interfaces Footers Padding( padding: const EdgeInsets.symmetric(vertical: 12.0, horizontal: 8.0), child: Row( mainAxisAlignment: MainAxisAlignment.spaceEvenly, children: [ OutlinedButton.icon( onPressed: _undo, icon: const Icon(Icons.undo), label: const Text('Undo'), ), ElevatedButton.icon( style: ElevatedButton.styleFrom(backgroundColor: Colors.redAccent.shade700), onPressed: () { Navigator.of(context).popUntil((route) => route.isFirst); }, icon: const Icon(Icons.refresh, color: Colors.white), label: const Text('New Game', style: TextStyle(color: Colors.white)), ), ], ), ) ], ), ); } }
